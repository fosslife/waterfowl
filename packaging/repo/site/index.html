Fosslife Packages

apt & dnf repository for Fosslife desktop apps (Waterfowl and more).

Debian / Ubuntu (apt)

# Add the signing key
curl -fsSL https://fosslife.github.io/packages/fosslife-packages.asc \
  | sudo gpg --dearmor -o /usr/share/keyrings/fosslife.gpg

# Add the repository
echo "deb [signed-by=/usr/share/keyrings/fosslife.gpg] https://fosslife.github.io/packages/deb ./" \
  | sudo tee /etc/apt/sources.list.d/fosslife.list

sudo apt update
sudo apt install waterfowl

Fedora / RHEL (dnf)

sudo tee /etc/yum.repos.d/fosslife.repo <<'EOF'
[fosslife]
name=Fosslife Packages
baseurl=https://fosslife.github.io/packages/rpm
enabled=1
gpgcheck=1
repo_gpgcheck=1
gpgkey=https://fosslife.github.io/packages/fosslife-packages.asc
EOF

sudo dnf install waterfowl
